// PRONTO ORB · SCANNER v2  (indicator)  —  één symbool per chart, 5m timeframe
//
// Scant elke combinatie van:
//   starttijd (elke 30 min, 00:00 → 23:59)  ×  ORB-lengte (10..30)
//   × VWAP (Met / Tegen)  ×  SL× (1..3)  ×  RR (1..3)  ×  VENSTER (60..300 min)
//
// Regels (identiek aan Multi-Slot Live v7):
//   - ORB = high/low van de bars in [start, start+ORB).
//   - Entry = eerste bar-close die de ORB doorbreekt (met VWAP-filter), binnen
//     het venster na het sluiten van de ORB. Eén trade per combinatie per dag.
//   - SL-afstand = SL× · (afstand entry → andere kant van de ORB).
//   - TP = RR · SL-afstand. SL en TP in dezelfde bar = verlies.
//   - Trade wordt pas gecontroleerd vanaf de bar NA de entry.
//   - Timeout: na max houdtijd op de close gesloten.
//
// NIEUW t.o.v. v1
//   - VENSTER is een scan-dimensie (standaard 60,120,180,240,300). Eén trade
//     telt mee voor elk venster dat groot genoeg is: geen extra rekentijd.
//   - EV in R (resultaat / SL-afstand). Bij vast risico-% per trade (zoals v7)
//     is dit de juiste maatstaf; EV in punten bevoordeelt brede stops.
//   - Modus "Uur-overzicht": per starttijd hoeveel combinaties positief zijn.
//   - Modus "Beste per starttijd (ranking)" (standaard): per starttijd (bv. elke 30 min)
//     alleen de beste instelling, gerangschikt van beste naar minste starttijd.
//   - Houdtijd standaard 2 dagen (2880 min). Kostenveld verwijderd.
//   - Pine Logs bevatten nu ook slWin (venster per slot).
// ═══════════════════════════════════════════════════════════════════════════

f_parse(string s) =>
    array<float> out = array.new<float>()
    array<string> parts = str.split(s, ",")
    for p in parts
        float v = str.tonumber(str.trim(p))
        if not na(v) and v > 0
            array.push(out, v)
    out

// id = (((start·nO + orb)·2 + vwap)·K + paar)·nW + venster
f_dec(int id) =>
    int wi  = id % nW
    int r1  = int((id - wi) / nW)
    int pj  = r1 % K
    int r2  = int((r1 - pj) / K)
    int vi  = r2 % 2
    int r3  = int((r2 - vi) / 2)
    int oi  = r3 % nO
    int si  = int((r3 - oi) / nO)
    [si, oi, vi, pj, wi]

f_hm(int m) =>
    str.format("{0,number,00}:{1,number,00}", math.floor(m / 60), m % 60)

f_row(table tb, int r, int id, string tag) =>
    [si, oi, vi, pj, wi] = f_dec(id)
    int stm   = array.get(starts, si)
    int n     = array.get(cN, id)
    int tp    = array.get(cW, id)
    float evp = n > 0 ? array.get(cS, id) / n : na
    float evr = n > 0 ? array.get(cR, id) / n : na
    float wr  = n > 0 ? tp / n * 100 : na
    float vr  = n > 0 ? math.max(array.get(cR2, id) / n - evr * evr, 0.0) : na
    float sd  = n > 1 ? math.sqrt(vr * n / (n - 1)) : na
    float tt  = (not na(sd) and sd > 0) ? evr / (sd / math.sqrt(n)) : na
    int tov   = array.get(cTo, id)
    float tor = n > 0 ? tov / n * 100 : na
    int no    = array.get(cNo, id)
    float evo = no > 0 ? array.get(cRo, id) / no : na
    color tc  = na(tt) ? color.gray : tt >= 2 ? color.lime : tt >= 1 ? color.yellow : color.silver
    color oc  = na(evo) ? color.gray : evo > 0 ? color.lime : color.red
    table.cell(tb, 0, r, tag, text_color = color.white, text_size = size.tiny)
    table.cell(tb, 1, r, f_hm(stm), text_color = color.white, text_size = size.tiny)
    table.cell(tb, 2, r, str.tostring(array.get(orbMin, oi)) + "m", text_color = color.white, text_size = size.tiny)
    table.cell(tb, 3, r, str.tostring(array.get(winL, wi)) + "m", text_color = color.aqua, text_size = size.tiny)
    table.cell(tb, 4, r, "×" + str.tostring(array.get(prM, pj), "#.#"), text_color = color.white, text_size = size.tiny)
    table.cell(tb, 5, r, vi == 0 ? "Met" : "Tegen", text_color = color.white, text_size = size.tiny)
    table.cell(tb, 6, r, str.tostring(array.get(prR, pj), "#.#") + "R", text_color = color.white, text_size = size.tiny)
    table.cell(tb, 7, r, str.tostring(n), text_color = color.white, text_size = size.tiny)
    table.cell(tb, 8, r, str.tostring(tp), text_color = color.white, text_size = size.tiny)
    table.cell(tb, 9, r, na(wr) ? "—" : str.tostring(wr, "0") + "%", text_color = color.white, text_size = size.tiny)
    table.cell(tb, 10, r, na(tor) ? "—" : str.tostring(tor, "0") + "%", text_color = (not na(tor) and tor >= 50) ? color.orange : color.white, text_size = size.tiny)
    table.cell(tb, 11, r, na(evp) ? "—" : str.tostring(evp, "0.00"), text_color = tc, text_size = size.tiny)
    table.cell(tb, 12, r, na(evr) ? "—" : str.tostring(evr, "0.00"), text_color = tc, text_size = size.tiny)
    table.cell(tb, 13, r, na(tt) ? "—" : str.tostring(tt, "0.0"), text_color = tc, text_size = size.tiny)
    table.cell(tb, 14, r, str.tostring(no), text_color = color.silver, text_size = size.tiny)
    table.cell(tb, 15, r, na(evo) ? "—" : str.tostring(evo, "0.00"), text_color = oc, text_size = size.tiny)
    n

// ── 1) Lopende trades afhandelen (vóór nieuwe entries: geen look-ahead) ───
if ok and barstate.isconfirmed and array.size(trades) > 0
    for ti = array.size(trades) - 1 to 0
        tr   = array.get(trades, ti)
        _fav = tr.dir == 1 ? high - tr.e : tr.e - low
        _adv = tr.dir == 1 ? tr.e - low : high - tr.e
        _mv  = tr.dir == 1 ? close - tr.e : tr.e - close
        _end = time >= tr.dl
        // snelle uitweg: raakt deze bar nog geen enkele SL/TP en is de houdtijd niet om, dan valt er niets te doen
        bool _act = _adv >= minSlF * tr.d or _fav >= minTpF * tr.d or _end
        if _act
            for pj = 0 to K - 1
                if array.get(tr.done, pj) == 0
                    _sd = array.get(prM, pj) * tr.d
                    _rr = array.get(prR, pj)
                    float _res = na
                    bool  _w   = false
                    bool  _to  = false
                    if _adv >= _sd
                        _res := -_sd
                    else if _fav >= _rr * _sd
                        _res := _rr * _sd
                        _w   := true
                    else if _end
                        _res := _mv
                        _to  := true
                    if not na(_res)
                        _net = _res
                        _rv  = _net / _sd
                        _b   = tr.base + pj * nW
                        for wi = tr.w0 to nW - 1
                            _id = _b + wi
                            if tr.oos
                                array.set(cNo, _id, array.get(cNo, _id) + 1)
                                array.set(cSo, _id, array.get(cSo, _id) + _net)
